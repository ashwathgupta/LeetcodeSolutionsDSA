class RideSharingSystem {

    private  Queue<Integer> qRider = new LinkedList<>();
    private  Queue<Integer> qDriver = new LinkedList<>();

    //if we use private static then leetcode will fail because multiple test cases run using the same class, so old data remains.

    public RideSharingSystem() {

    }

    public void addRider(int riderId) {
        qRider.offer(riderId);

    }

    public void addDriver(int driverId) {
        qDriver.offer(driverId);

    }

    public int[] matchDriverWithRider() {

        if (qRider.isEmpty() || qDriver.isEmpty()) {
            return new int[] { -1, -1 };
        }
        return new int[] { qDriver.poll(), qRider.poll() };

    }

    public void cancelRider(int riderId) {
        if (qRider.isEmpty()) {
            return;
        }
        qRider.removeIf(x -> x == riderId);
        // qRider = qRider.stream().filter(x -> x != riderId).collect(Collectors.toCollection(LinkedList::new));
        return;
    }
}
